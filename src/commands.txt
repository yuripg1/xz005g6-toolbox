# Plaintext sys commands. One per line. Format: LABEL | COMMAND
# Lines starting with # are ignored. Blank lines are ignored.
# The 'warmup' command MUST come first — it consumes the first-command key state.

warmup     | linuxsh
tftp_get   | linuxsh tftp -g -l /tmp/new.squashfs -r patched_rootfs.squashfs __HOST_IP__ __TFTP_PORT__
check_size | linuxsh ls -la /tmp/new.squashfs
check_md5  | linuxsh md5sum /tmp/new.squashfs
backup     | linuxsh cat /dev/mtd5 > /tmp/old_rootfs_backup.squashfs
backup_ok  | linuxsh ls -la /tmp/old_rootfs_backup.squashfs
erase      | linuxsh flash_eraseall /dev/mtd5
write      | linuxsh cat /tmp/new.squashfs > /dev/mtd5
verify     | linuxsh cmp /tmp/new.squashfs /dev/mtd5
reboot     | linuxsh reboot
check_bdp  | linuxsh omcicli get cflag
check_omci | linuxsh flash get OMCI_CUSTOM_BDP
check_dual | linuxsh flash get DUAL_MGMT_MODE
